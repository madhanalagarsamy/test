<!DOCTYPE html>
<html>
<head>
    <title>Image Map</title>
</head>

<body>

    <h2 align="center">Image Map Example</h2>

    <p align="center">
        Click on the Laptop, Mobile, Printer or Coffee Cup
    </p>

    <img src="workspace.jpg"
         alt="Workspace"
         usemap="#map1"
         width="500"
         height="300">

    <map name="map1">

        <!-- Laptop -->
        <area shape="rect"
              coords="100,80,300,220"
              href="https://www.youtube.com"
              alt="Laptop">

        <!-- Mobile -->
        <area shape="rect"
              coords="320,80,380,220"
              href="https://www.yahoo.com"
              alt="Mobile">

        <!-- Printer -->
        <area shape="rect"
              coords="50,220,180,290"
              href="https://www.google.com"
              alt="Printer">

        <!-- Coffee Cup -->
        <area shape="circle"
              coords="420,230,35"
              href="https://www.wikipedia.org"
              alt="Coffee Cup">

    </map>

</body>
