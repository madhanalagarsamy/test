Create a Web Page and Add JavaScript to Handle Mouse Events and Form Events




index.html


<!DOCTYPE html>
<html>

<head>

    <title>Mouse and Form Events</title>

    <style>
        #box {
            width: 300px;
            height: 120px;
            background: lightblue;
            text-align: center;
            line-height: 120px;
            border: 2px solid black;
            font-size: 20px;
            margin-bottom: 20px;
        }
    </style>

    <script>

        function mouseOver() {
            document.getElementById("box").innerHTML = "Mouse Over";
            document.getElementById("box").style.background = "yellow";
        }

        function mouseOut() {
            document.getElementById("box").innerHTML = "Mouse Out";
            document.getElementById("box").style.background = "lightblue";
        }

        function mouseClick() {
            document.getElementById("box").innerHTML = "Single Click";
            document.getElementById("box").style.background = "lightgreen";
        }

        function doubleClick() {
            document.getElementById("box").innerHTML = "Double Click";
            document.getElementById("box").style.background = "orange";
        }

        function mouseDown() {
            document.getElementById("box").innerHTML = "Mouse Button Pressed";
        }

        function mouseUp() {
            document.getElementById("box").innerHTML = "Mouse Button Released";
        }

        function validateForm() {

            var name = document.getElementById("name").value;

            if (name == "") {
                alert("Please enter your name");
                return false;
            }

            document.getElementById("msg").innerHTML =
                "Form Submitted Successfully";

            return false;
        }

        function focusField() {
            document.getElementById("name").style.background = "pink";
        }

        function blurField() {
            document.getElementById("name").style.background = "white";
        }

    </script>

</head>

<body>

    <h2>Mouse Events Demo</h2>

    <div id="box"
         onmouseover="mouseOver()"
         onmouseout="mouseOut()"
         onclick="mouseClick()"
         ondblclick="doubleClick()"
         onmousedown="mouseDown()"
         onmouseup="mouseUp()">

        Perform Mouse Events Here

    </div>


    <h2>Form Events Demo</h2>

    <form onsubmit="return validateForm()">

        Name:

        <input type="text"
               id="name"
               onfocus="focusField()"
               onblur="blurField()">

        <br><br>

        <input type="submit" value="Submit">

        <p id="msg"
           style="color:green; font-weight:bold;">
        </p>

    </form>

</body>

</html>
