Web Page to Demonstrate Inline CSS, Internal CSS and External CSS

<!DOCTYPE html>
<html>
<head>
    <title>CSS Demonstration</title>

    <!-- External CSS -->
    <link rel="stylesheet" href="style.css">

    <!-- Internal CSS -->
    <style>
        h2 {
            color: blue;
            text-align: center;
        }

        p {
            font-size: 18px;
            color: green;
        }
    </style>
</head>

<body>

    <!-- Inline CSS -->
    <h1 style="color: red; text-align: center;">
        Welcome to CSS Demonstration
    </h1>

    <!-- Internal CSS -->
    <h2>This heading uses Internal CSS</h2>

    <p>This paragraph is styled using Internal CSS</p>

    <!-- External CSS -->
    <div class="box">
        This box is styled using External CSS
    </div>

</body>
</html>
